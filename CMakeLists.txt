cmake_minimum_required(VERSION 3.21)
project(DigitalWindTunnel VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.4)
qt_policy(SET QTP0001 NEW)
qt_policy(SET QTP0004 NEW)
if(NOT VTK_DIR)
  file(GLOB DWT_VTK_CONFIGS "$ENV{HOME}/.local/lib/cmake/vtk-*/vtk-config.cmake")
  list(SORT DWT_VTK_CONFIGS COMPARE NATURAL ORDER DESCENDING)
  if(DWT_VTK_CONFIGS)
    list(GET DWT_VTK_CONFIGS 0 DWT_VTK_CONFIG)
    get_filename_component(VTK_DIR "${DWT_VTK_CONFIG}" DIRECTORY)
    set(VTK_DIR "${VTK_DIR}" CACHE PATH "VTK package directory")
  endif()
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/inja/CMakeLists.txt")
  message(FATAL_ERROR "inja submodule missing; run: git submodule update --init --recursive")
endif()
set(INJA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(INJA_INSTALL OFF CACHE BOOL "" FORCE)
set(INJA_EXPORT OFF CACHE BOOL "" FORCE)
add_subdirectory(external/inja EXCLUDE_FROM_ALL)
find_package(VTK CONFIG QUIET COMPONENTS CommonCore CommonDataModel FiltersCore IOGeometry IOXML GUISupportQtQuick)
qt_add_executable(digital-wind-tunnel src/main.cpp src/SimulationController.cpp src/SimulationController.h src/OpenFoamCase.cpp src/OpenFoamCase.h src/VtkView.cpp src/VtkView.h)
qt_add_qml_module(digital-wind-tunnel URI WindTunnel VERSION 1.0 QML_FILES qml/Main.qml qml/MetricCard.qml qml/ResidualPlot.qml)
file(GLOB_RECURSE DWT_CASE_TEMPLATES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" templates/windTunnel/*)
list(FILTER DWT_CASE_TEMPLATES EXCLUDE REGEX "\\.md$")
qt_add_resources(digital-wind-tunnel case_templates PREFIX "/" FILES ${DWT_CASE_TEMPLATES})
target_include_directories(digital-wind-tunnel PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(digital-wind-tunnel PRIVATE Qt6::Quick Qt6::QuickControls2 pantor::inja)
if(VTK_FOUND)
  target_compile_definitions(digital-wind-tunnel PRIVATE WINDTUNNEL_HAS_VTK=1)
  target_link_libraries(digital-wind-tunnel PRIVATE ${VTK_LIBRARIES})
  vtk_module_autoinit(TARGETS digital-wind-tunnel MODULES ${VTK_LIBRARIES})
else()
  message(STATUS "VTK Qt Quick support not found; building without the 3D results view")
endif()
install(TARGETS digital-wind-tunnel RUNTIME DESTINATION bin)
